<script>
    function agregarFila() {
        var contenedor = document.getElementById("contenedorProductos");
        var filaOriginal = document.querySelector(".producto-fila");
        var nuevaFila = filaOriginal.cloneNode(true);
        nuevaFila.querySelector("input").value = 1;
        contenedor.appendChild(nuevaFila);
        calcularTotal();
    }

    function removerFila(btn) {
        var filas = document.querySelectorAll(".producto-fila");
        if(filas.length > 1) {
            btn.closest(".producto-fila").remove();
            calcularTotal();
        }
    }

    function calcularTotal() {
        var total = 0;
        var filas = document.querySelectorAll(".producto-fila");
        filas.forEach(function(fila) {
            var select = fila.querySelector(".prod-select");
            var cantidadInput = fila.querySelector(".prod-cant");
            var optionSelected = select.options[select.selectedIndex];
            var precio = parseFloat(optionSelected.getAttribute("data-precio")) || 0;
            var cantidad = parseInt(cantidadInput.value) || 0;
